cmake_minimum_required(VERSION 3.22)
# Must precede project(). Will's Mac: M1 Max, macOS 26.6.
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum macOS version")
project(FOHGauge VERSION 0.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FOH_BUILD_TESTS "Build core unit tests" ON)
option(FOH_BUILD_APP "Build JUCE app" ON)
set(FOH_SANITIZE "" CACHE STRING "Opt-in core/app/test sanitizers (address,undefined)")
option(FOH_TSAN "Instrument core, app and tests with ThreadSanitizer" OFF)
add_library(foh_sanitizers INTERFACE)
if(FOH_TSAN)
  if(FOH_SANITIZE)
    message(FATAL_ERROR "FOH_TSAN requires a separate build from FOH_SANITIZE")
  endif()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(AppleClang|Clang|GNU)$")
    message(FATAL_ERROR "FOH_TSAN requires Clang or GCC")
  endif()
  target_compile_options(foh_sanitizers INTERFACE -fsanitize=thread -fno-omit-frame-pointer -g)
  target_link_options(foh_sanitizers INTERFACE -fsanitize=thread)
endif()
if(FOH_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(AppleClang|Clang|GNU)$")
    message(FATAL_ERROR "FOH_SANITIZE requires Clang or GCC")
  endif()
  if(NOT FOH_SANITIZE MATCHES "^(address|undefined)(,(address|undefined))*$")
    message(FATAL_ERROR "FOH_SANITIZE must be a comma-separated list of address and/or undefined")
  endif()
  target_compile_options(foh_sanitizers INTERFACE
    "-fsanitize=${FOH_SANITIZE}" -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(foh_sanitizers INTERFACE "-fsanitize=${FOH_SANITIZE}")
endif()
include(FetchContent)
add_subdirectory(core)
if(FOH_BUILD_TESTS)
  FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(Catch2)
  enable_testing()
  add_subdirectory(tests)
endif()
if(FOH_BUILD_APP)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
  add_subdirectory(app)
endif()
